/**
 * Buscar productos por código, nombre o SKU
 * @param string $termino Término de búsqueda
 * @param int $limite Cantidad máxima de resultados
 * @return array Lista de productos encontrados
 */
public function buscar($termino, $limite = 20) {
    try {
        $sql = "SELECT 
                    p.id_producto,
                    p.codigo,
                    p.sku,
                    p.nombre,
                    p.descripcion,
                    p.stock_actual,
                    p.stock_minimo,
                    p.stock_maximo,
                    p.precio_compra,
                    p.precio_venta,
                    p.estado_stock,
                    c.nombre AS categoria_nombre,
                    m.nombre AS marca_nombre
                FROM productos p
                LEFT JOIN categorias c ON p.id_categoria = c.id_categoria
                LEFT JOIN marcas m ON p.id_marca = m.id_marca
                WHERE p.estado = 'activo'
                    AND (
                        p.codigo LIKE ? 
                        OR p.nombre LIKE ? 
                        OR p.sku LIKE ?
                        OR p.descripcion LIKE ?
                    )
                ORDER BY p.nombre ASC
                LIMIT ?";
        
        $terminoBusqueda = "%{$termino}%";
        $stmt = $this->db->prepare($sql);
        $stmt->execute([
            $terminoBusqueda,
            $terminoBusqueda,
            $terminoBusqueda,
            $terminoBusqueda,
            $limite
        ]);
        
        return $stmt->fetchAll(PDO::FETCH_ASSOC);
        
    } catch (PDOException $e) {
        error_log("Error en buscar: " . $e->getMessage());
        return [];
    }
}

// MÉTODO 2: Obtener productos con stock bajo
// ============================================
/**
 * Obtener productos con stock bajo o crítico
 * @return array Lista de productos con alerta de stock
 */
public function getProductosStockBajo() {
    try {
        $sql = "SELECT 
                    p.*,
                    c.nombre AS categoria_nombre,
                    m.nombre AS marca_nombre,
                    (p.stock_maximo - p.stock_actual) AS cantidad_requerir,
                    ((p.stock_maximo - p.stock_actual) * p.precio_compra) AS monto_requerido
                FROM productos p
                LEFT JOIN categorias c ON p.id_categoria = c.id_categoria
                LEFT JOIN marcas m ON p.id_marca = m.id_marca
                WHERE p.estado = 'activo'
                    AND p.stock_actual <= p.stock_minimo
                ORDER BY p.estado_stock DESC, p.stock_actual ASC";
        
        $stmt = $this->db->prepare($sql);
        $stmt->execute();
        
        return $stmt->fetchAll(PDO::FETCH_ASSOC);
        
    } catch (PDOException $e) {
        error_log("Error en getProductosStockBajo: " . $e->getMessage());
        return [];
    }
}

/**
 * Actualizar el stock de un producto
 * @param int $idProducto ID del producto
 * @param int $nuevoStock Nuevo valor de stock
 * @return bool True si se actualizó correctamente
 */
public function actualizarStock($idProducto, $nuevoStock) {
    try {
        $this->db->beginTransaction();
        
        // Obtener stock anterior
        $sql = "SELECT stock_actual FROM productos WHERE id_producto = ?";
        $stmt = $this->db->prepare($sql);
        $stmt->execute([$idProducto]);
        $stockAnterior = $stmt->fetchColumn();
        
        // Actualizar stock
        $sql = "UPDATE productos 
                SET stock_actual = ?,
                    estado_stock = CASE 
                        WHEN ? = 0 THEN 'agotado'
                        WHEN ? < (stock_minimo * 0.5) THEN 'critico'
                        WHEN ? <= stock_minimo THEN 'bajo'
                        ELSE 'normal'
                    END,
                    fecha_actualizacion = NOW()
                WHERE id_producto = ?";
        
        $stmt = $this->db->prepare($sql);
        $stmt->execute([
            $nuevoStock,
            $nuevoStock,
            $nuevoStock,
            $nuevoStock,
            $idProducto
        ]);
        
        $this->db->commit();
        return true;
        
    } catch (PDOException $e) {
        $this->db->rollBack();
        error_log("Error en actualizarStock: " . $e->getMessage());
        return false;
    }
}

// MÉTODO 4: Obtener historial de precios
// ============================================
/**
 * Obtener historial de precios del producto desde el kardex
 * @param int $idProducto ID del producto
 * @param int $limite Cantidad de registros a obtener
 * @return array Historial de precios
 */
public function getHistorialPrecios($idProducto, $limite = 10) {
    try {
        $sql = "SELECT 
                    k.fecha_movimiento,
                    k.tipo_movimiento,
                    CASE 
                        WHEN k.tipo_movimiento = 'entrada' THEN k.entrada_costo_unitario
                        WHEN k.tipo_movimiento = 'salida' THEN k.salida_costo_unitario
                    END AS precio_unitario,
                    k.documento_referencia,
                    mi.observaciones
                FROM kardex k
                INNER JOIN movimientos_inventario mi ON k.id_movimiento = mi.id_movimiento
                WHERE k.id_producto = ?
                    AND (k.entrada_costo_unitario > 0 OR k.salida_costo_unitario > 0)
                ORDER BY k.fecha_movimiento DESC
                LIMIT ?";
        
        $stmt = $this->db->prepare($sql);
        $stmt->execute([$idProducto, $limite]);
        
        return $stmt->fetchAll(PDO::FETCH_ASSOC);
        
    } catch (PDOException $e) {
        error_log("Error en getHistorialPrecios: " . $e->getMessage());
        return [];
    }
}

// MÉTODO 5: Validar stock disponible
// ============================================
/**
 * Validar si hay stock disponible para una salida
 * @param int $idProducto ID del producto
 * @param int $cantidadSolicitada Cantidad que se desea sacar
 * @return array ['disponible' => bool, 'stock_actual' => int, 'mensaje' => string]
 */
public function validarStockDisponible($idProducto, $cantidadSolicitada) {
    try {
        $sql = "SELECT stock_actual, nombre FROM productos WHERE id_producto = ?";
        $stmt = $this->db->prepare($sql);
        $stmt->execute([$idProducto]);
        $producto = $stmt->fetch(PDO::FETCH_ASSOC);
        
        if (!$producto) {
            return [
                'disponible' => false,
                'stock_actual' => 0,
                'mensaje' => 'Producto no encontrado'
            ];
        }
        
        $stockActual = $producto['stock_actual'];
        
        if ($stockActual < $cantidadSolicitada) {
            return [
                'disponible' => false,
                'stock_actual' => $stockActual,
                'mensaje' => "Stock insuficiente. Disponible: {$stockActual}, Solicitado: {$cantidadSolicitada}"
            ];
        }
        
        return [
            'disponible' => true,
            'stock_actual' => $stockActual,
            'mensaje' => 'Stock disponible'
        ];
        
    } catch (PDOException $e) {
        error_log("Error en validarStockDisponible: " . $e->getMessage());
        return [
            'disponible' => false,
            'stock_actual' => 0,
            'mensaje' => 'Error al validar stock'
        ];
    }
}

// MÉTODO 6: Obtener productos más vendidos
// ============================================
/**
 * Obtener los productos más vendidos en un período
 * @param string $fechaInicio Fecha inicial
 * @param string $fechaFin Fecha final
 * @param int $limite Cantidad de productos
 * @return array Lista de productos más vendidos
 */
public function getProductosMasVendidos($fechaInicio, $fechaFin, $limite = 10) {
    try {
        $sql = "SELECT 
                    p.id_producto,
                    p.codigo,
                    p.nombre,
                    p.stock_actual,
                    c.nombre AS categoria_nombre,
                    SUM(k.salida_cantidad) AS total_vendido,
                    SUM(k.salida_costo_total) AS valor_total_ventas,
                    COUNT(DISTINCT k.id_kardex) AS numero_ventas
                FROM productos p
                INNER JOIN kardex k ON p.id_producto = k.id_producto
                LEFT JOIN categorias c ON p.id_categoria = c.id_categoria
                WHERE k.tipo_movimiento = 'salida'
                    AND k.fecha_movimiento BETWEEN ? AND ?
                    AND p.estado = 'activo'
                GROUP BY p.id_producto, p.codigo, p.nombre, p.stock_actual, c.nombre
                ORDER BY total_vendido DESC
                LIMIT ?";
        
        $stmt = $this->db->prepare($sql);
        $stmt->execute([$fechaInicio, $fechaFin, $limite]);
        
        return $stmt->fetchAll(PDO::FETCH_ASSOC);
        
    } catch (PDOException $e) {
        error_log("Error en getProductosMasVendidos: " . $e->getMessage());
        return [];
    }
}

// MÉTODO 7: Calcular rotación de inventario
// ============================================
/**
 * Calcular el índice de rotación de un producto
 * @param int $idProducto ID del producto
 * @param string $fechaInicio Fecha inicial
 * @param string $fechaFin Fecha final
 * @return array Datos de rotación
 */
public function calcularRotacion($idProducto, $fechaInicio, $fechaFin) {
    try {
        $sql = "SELECT 
                    p.nombre,
                    p.stock_actual,
                    COALESCE(SUM(k.salida_cantidad), 0) AS total_salidas,
                    COALESCE(SUM(k.entrada_cantidad), 0) AS total_entradas,
                    CASE 
                        WHEN p.stock_actual > 0 
                        THEN ROUND(COALESCE(SUM(k.salida_cantidad), 0) / p.stock_actual, 2)
                        ELSE 0
                    END AS indice_rotacion,
                    CASE 
                        WHEN COALESCE(SUM(k.salida_cantidad), 0) > p.stock_actual * 3 THEN 'Alta'
                        WHEN COALESCE(SUM(k.salida_cantidad), 0) > p.stock_actual THEN 'Media'
                        ELSE 'Baja'
                    END AS clasificacion_rotacion,
                    DATEDIFF(?, ?) AS dias_periodo
                FROM productos p
                LEFT JOIN kardex k ON p.id_producto = k.id_producto 
                    AND k.fecha_movimiento BETWEEN ? AND ?
                WHERE p.id_producto = ?
                GROUP BY p.id_producto, p.nombre, p.stock_actual";
        
        $stmt = $this->db->prepare($sql);
        $stmt->execute([
            $fechaFin, 
            $fechaInicio, 
            $fechaInicio, 
            $fechaFin, 
            $idProducto
        ]);
        
        $rotacion = $stmt->fetch(PDO::FETCH_ASSOC);
        
        // Calcular rotación mensual
        if ($rotacion && $rotacion['dias_periodo'] > 0) {
            $rotacion['rotacion_mensual'] = round(
                ($rotacion['indice_rotacion'] * 30) / $rotacion['dias_periodo'], 
                2
            );
        }
        
        return $rotacion;
        
    } catch (PDOException $e) {
        error_log("Error en calcularRotacion: " . $e->getMessage());
        return null;
    }
}

// MÉTODO 8: Generar código de producto automático
// ============================================
/**
 * Generar el siguiente código de producto disponible
 * @param string $prefijo Prefijo del código (ej: 'PROD-')
 * @return string Nuevo código generado
 */
public function generarCodigoProducto($prefijo = 'PROD-') {
    try {
        $sql = "SELECT codigo 
                FROM productos 
                WHERE codigo LIKE ? 
                ORDER BY codigo DESC 
                LIMIT 1";
        
        $stmt = $this->db->prepare($sql);
        $stmt->execute([$prefijo . '%']);
        $ultimoCodigo = $stmt->fetchColumn();
        
        if ($ultimoCodigo) {
            // Extraer el número del último código
            $numero = (int)str_replace($prefijo, '', $ultimoCodigo);
            $nuevoNumero = $numero + 1;
        } else {
            $nuevoNumero = 1;
        }
        
        // Formatear con ceros a la izquierda
        return $prefijo . str_pad($nuevoNumero, 6, '0', STR_PAD_LEFT);
        
    } catch (PDOException $e) {
        error_log("Error en generarCodigoProducto: " . $e->getMessage());
        return $prefijo . date('YmdHis'); // Código alternativo basado en fecha/hora
    }
}
